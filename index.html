import sys import os import pandas as pd from PyQt6.QtWidgets import (QApplication, QMainWindow, QWidget, QVBoxLayout, QHBoxLayout, QLineEdit, QScrollArea, QLabel, QFrame, QPushButton, QTabWidget, QGridLayout) from PyQt6.QtGui import QPixmap, QFont from PyQt6.QtCore import Qt # 사진이 들어있는 메인 폴더 경로 (필요에 따라 수정) PHOTO_BASE_PATH = "./photos" class PersonCard(QFrame): def __init__(self, data): super().__init__() self.data = data self.setFrameShape(QFrame.Shape.StyledPanel) self.setLineWidth(2) self.setStyleSheet("background-color: #ffffff; border-radius: 10px; margin-bottom: 20px;") self.init_ui() def init_ui(self): main_layout = QVBoxLayout(self) # --- 상단 정보 (이름, 연락처, 생년월일, 소속) --- top_layout = QHBoxLayout() name_label = QLabel(str(self.data.get('이름', ''))) name_label.setFont(QFont("Arial", 16, QFont.Weight.Bold)) phone_label = QLabel(str(self.data.get('연락처', ''))) birth_label = QLabel(str(self.data.get('생년월일', ''))) group_label = QLabel(str(self.data.get('소속', ''))) top_layout.addWidget(name_label) top_layout.addWidget(phone_label) top_layout.addStretch() top_layout.addWidget(birth_label) top_layout.addStretch() top_layout.addWidget(group_label) main_layout.addLayout(top_layout) # --- 사진 탭 시스템 --- self.tabs = QTabWidget() self.load_photos(str(self.data.get('이름', ''))) main_layout.addWidget(self.tabs) # --- 상세 정보 섹션 --- info_layout = QGridLayout() # 섹션 정의 및 데이터 매핑 sections = { "1. 기본정보": ["주소", "보유자료", "사이즈", "관찰동기", "첫만남", "첫스킨십", "첫빠구리"], "2. 관계 상태/진행도": ["진척도", "나에대한 호감도", "최근수위", "최근고민"], "3. 성향/취향": ["선호취향", "특이사항", "특이반응"], "4. 리스크/주의사항": ["주의사항", "경계인물"], "5. 접근성/운영정보": ["접근가능시간", "누적지출", "가성비"], "6. 난이도/전략": ["공략난이도", "공략계획"] } row = 0 for title, fields in sections.items(): title_lbl = QLabel(title) title_lbl.setFont(QFont("Arial", 10, QFont.Weight.Bold)) info_layout.addWidget(title_lbl, row, 0, 1, 2) row += 1 for field in fields: val = str(self.data.get(field, '')) lbl = QLabel(f"• {field}: {val}") info_layout.addWidget(lbl, row, 0, 1, 2) row += 1 # 7. 시스템 정보 (노션링크 버튼) notion_btn = QPushButton("노션링크") notion_url = str(self.data.get('노션링크', '')) notion_btn.clicked.connect(lambda: os.system(f"start {notion_url}")) info_layout.addWidget(QLabel("7. 시스템정보"), row, 0) info_layout.addWidget(notion_btn, row, 1) main_layout.addLayout(info_layout) # 최하단 마지막 본 날 last_date = QLabel(f"마지막 본 날: {str(self.data.get('마지막 본날짜', ''))}") last_date.setStyleSheet("font-size: 9px; color: gray;") main_layout.addWidget(last_date, alignment=Qt.AlignmentFlag.AlignRight) def load_photos(self, person_name): person_path = os.path.join(PHOTO_BASE_PATH, person_name) # 전체 탭 생성 all_tab = QScrollArea() all_widget = QWidget() all_layout = QHBoxLayout(all_widget) if os.path.exists(person_path): subfolders = [f for f in os.listdir(person_path) if os.path.isdir(os.path.join(person_path, f))] for folder in subfolders: folder_path = os.path.join(person_path, folder) sub_tab = QScrollArea() sub_widget = QWidget() sub_layout = QHBoxLayout(sub_widget) for img_file in os.listdir(folder_path): if img_file.lower().endswith(('.png', '.jpg', '.jpeg')): img_path = os.path.join(folder_path, img_file) # 개별 탭용 sub_layout.addWidget(self.create_img_label(img_path)) # 전체 탭용 all_layout.addWidget(self.create_img_label(img_path)) sub_tab.setWidget(sub_widget) sub_tab.setWidgetResizable(True) self.tabs.addTab(sub_tab, folder) all_tab.setWidget(all_widget) all_tab.setWidgetResizable(True) self.tabs.insertTab(0, all_tab, "전체") def create_img_label(self, path): lbl = QLabel() pixmap = QPixmap(path).scaled(200, 200, Qt.AspectRatioMode.KeepAspectRatio) lbl.setPixmap(pixmap) lbl.setStyleSheet("border: 1px solid #ccc; padding: 5px;") return lbl class MainWindow(QMainWindow): def __init__(self): super().__init__() self.setWindowTitle("인원 관리 시스템") self.setGeometry(100, 100, 900, 800) self.central_widget = QWidget() self.setCentralWidget(self.central_widget) self.layout = QVBoxLayout(self.central_widget) # 검색창 self.search_input = QLineEdit() self.search_input.setPlaceholderText("이름 또는 소속으로 검색하세요...") self.search_input.textChanged.connect(self.filter_cards) self.layout.addWidget(self.search_input) # 스크롤 영역 self.scroll = QScrollArea() self.container = QWidget() self.card_layout = QVBoxLayout(self.container) self.scroll.setWidget(self.container) self.scroll.setWidgetResizable(True) self.layout.addWidget(self.scroll) self.cards = [] self.load_data() def load_data(self): try: # CSV 로드 (탭 구분자나 쉼표 확인 필요, 여기선 제공된 파일 구조에 맞춤) df = pd.read_csv('인명.csv', sep='\t') for _, row in df.iterrows(): card = PersonCard(row.to_dict()) self.card_layout.addWidget(card) self.cards.append(card) except Exception as e: print(f"데이터 로드 오류: {e}") def filter_cards(self): text = self.search_input.text().lower() for card in self.cards: name = str(card.data.get('이름', '')).lower() group = str(card.data.get('소속', '')).lower() card.setVisible(text in name or text in group) if __name__ == "__main__": app = QApplication(sys.argv) window = MainWindow() window.show() sys.exit(app.exec())
