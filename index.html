import sys import os import pandas as pd import configparser from PyQt6.QtWidgets import (QApplication, QMainWindow, QWidget, QVBoxLayout, QHBoxLayout, QLineEdit, QScrollArea, QLabel, QFrame, QPushButton, QTabWidget, QGridLayout, QInputDialog, QMessageBox) from PyQt6.QtGui import QPixmap, QFont from PyQt6.QtCore import Qt # 설정 파일 관리 (비밀번호 저장용) CONFIG_FILE = 'config.ini' PHOTO_BASE_PATH = "./photos" # 사진 폴더 경로 class LoginDialog(QInputDialog): def __init__(self): super().__init__() self.config = configparser.ConfigParser() self.load_config() def load_config(self): if not os.path.exists(CONFIG_FILE): self.config['SETTINGS'] = {'password': '0000'} with open(CONFIG_FILE, 'w') as f: self.config.write(f) else: self.config.read(CONFIG_FILE) def verify_password(self): pwd, ok = QInputDialog.getText(None, "보안 접속", "비밀번호 4자리를 입력하세요:", QLineEdit.EchoMode.Password) if ok: if pwd == self.config['SETTINGS']['password']: return True else: QMessageBox.critical(None, "오류", "비밀번호가 일치하지 않습니다.") return False return False class PersonCard(QFrame): def __init__(self, data): super().__init__() self.data = data self.setFrameShape(QFrame.Shape.StyledPanel) self.setStyleSheet("background-color: #ffffff; border: 1px solid #ddd; border-radius: 8px; margin-bottom: 15px;") self.init_ui() def init_ui(self): layout = QVBoxLayout(self) # 상단 헤더 (이름, 연락처, 생년월일, 소속) header = QHBoxLayout() name_lbl = QLabel(str(self.data.get('이름', ''))) name_lbl.setFont(QFont("Malgun Gothic", 15, QFont.Weight.Bold)) header.addWidget(name_lbl) header.addWidget(QLabel(str(self.data.get('연락처', '')))) header.addStretch() header.addWidget(QLabel(str(self.data.get('생년월일', '')))) header.addWidget(QLabel(f"[{str(self.data.get('소속', ''))}]")) layout.addLayout(header) # 사진 탭 self.tabs = QTabWidget() self.load_photos(str(self.data.get('이름', ''))) layout.addWidget(self.tabs) # 상세 항목 배치 grid = QGridLayout() sections = { "1. 기본정보": ["주소", "보유자료", "사이즈", "관찰동기", "첫만남", "첫스킨십", "첫빠구리"], "2. 관계 상태/진행도": ["진척도", "나에대한 호감도", "최근수위", "최근고민"], "3. 성향/취향": ["선호취향", "특이사항", "특이반응"], "4. 리스크/주의사항": ["주의사항", "경계인물"], "5. 접근성/운영정보": ["접근가능시간", "누적지출", "가성비"], "6. 난이도/전략": ["공략난이도", "공략계획"] } row = 0 for section_title, fields in sections.items(): title_lbl = QLabel(section_title) title_lbl.setStyleSheet("font-weight: bold; color: #2c3e50; margin-top: 5px;") grid.addWidget(title_lbl, row, 0, 1, 2) row += 1 for field in fields: val = str(self.data.get(field, '')) grid.addWidget(QLabel(f"• {field}: {val}"), row, 0, 1, 2) row += 1 # 7. 시스템 정보 notion_url = str(self.data.get('노션링크', '')) btn_notion = QPushButton("노션링크") btn_notion.clicked.connect(lambda: os.system(f"start {notion_url}")) grid.addWidget(QLabel("7. 시스템정보"), row, 0) grid.addWidget(btn_notion, row, 1) layout.addLayout(grid) # 마지막 본 날 footer = QLabel(f"마지막 본 날: {str(self.data.get('마지막 본날짜', ''))}") footer.setStyleSheet("font-size: 10px; color: #7f8c8d;") layout.addWidget(footer, alignment=Qt.AlignmentFlag.AlignRight) def load_photos(self, name): p_path = os.path.join(PHOTO_BASE_PATH, name) all_scroll = QScrollArea() all_widget = QWidget() all_lay = QHBoxLayout(all_widget) if os.path.exists(p_path): for folder in os.listdir(p_path): f_path = os.path.join(p_path, folder) if os.path.isdir(f_path): sub_scroll = QScrollArea() sub_widget = QWidget() sub_lay = QHBoxLayout(sub_widget) for img in os.listdir(f_path): if img.lower().endswith(('.jpg', '.png', '.jpeg')): path = os.path.join(f_path, img) sub_lay.addWidget(self.make_img(path)) all_lay.addWidget(self.make_img(path)) sub_scroll.setWidget(sub_widget) self.tabs.addTab(sub_scroll, folder) all_scroll.setWidget(all_widget) self.tabs.insertTab(0, all_scroll, "전체") def make_img(self, path): l = QLabel() l.setPixmap(QPixmap(path).scaled(180, 180, Qt.AspectRatioMode.KeepAspectRatio)) return l class MainApp(QMainWindow): def __init__(self): super().__init__() self.setWindowTitle("인원 관리 보안 시스템") self.setGeometry(100, 100, 1000, 900) main_layout = QVBoxLayout() # 최상단 메뉴 (검색 및 비번변경) top_bar = QHBoxLayout() self.search_bar = QLineEdit() self.search_bar.setPlaceholderText("이름 또는 소속 검색...") self.search_bar.textChanged.connect(self.filter_data) self.btn_change_pwd = QPushButton("비밀번호 변경") self.btn_change_pwd.clicked.connect(self.change_password) top_bar.addWidget(self.search_bar) top_bar.addWidget(self.btn_change_pwd) main_layout.addLayout(top_bar) # 카드 리스트 영역 self.scroll = QScrollArea() self.container = QWidget() self.card_list = QVBoxLayout(self.container) self.scroll.setWidget(self.container) self.scroll.setWidgetResizable(True) main_layout.addWidget(self.scroll) central = QWidget() central.setLayout(main_layout) self.setCentralWidget(central) self.cards = [] self.refresh_data() def refresh_data(self): if os.path.exists('인명.csv'): df = pd.read_csv('인명.csv', sep='\t') for _, row in df.iterrows(): card = PersonCard(row.to_dict()) self.card_list.addWidget(card) self.cards.append(card) def filter_data(self): t = self.search_bar.text().lower() for c in self.cards: match = t in str(c.data.get('이름','')).lower() or t in str(c.data.get('소속','')).lower() c.setVisible(match) def change_password(self): new_pwd, ok = QInputDialog.getText(self, "비밀번호 변경", "새로운 4자리 비밀번호를 입력하세요:") if ok and len(new_pwd) == 4 and new_pwd.isdigit(): config = configparser.ConfigParser() config['SETTINGS'] = {'password': new_pwd} with open(CONFIG_FILE, 'w') as f: config.write(f) QMessageBox.information(self, "완료", "비밀번호가 변경되었습니다.") else: QMessageBox.warning(self, "오류", "숫자 4자리를 입력해주세요.") if __name__ == "__main__": app = QApplication(sys.argv) # 보안 체크 login = LoginDialog() if login.verify_password(): ex = MainApp() ex.show() sys.exit(app.exec()) else: sys.exit()
